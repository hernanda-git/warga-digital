## Overview

Phase 2 builds the full-page `ArticleComposer` — replacing the modal-based `ArticleFormModal` with a WordPress Gutenberg–style single-page composer. The composer is accessible at `/admin/articles/compose` and supports both new article creation and editing of existing articles via URL parameter.

**New file:** `src/app/admin/articles/compose/page.tsx`

**Behavior:**
- Route: `/admin/articles/compose` — new article
- Route: `/admin/articles/compose?id={articleId}` — edit existing article

## 2.3 Load Existing Article (Edit Mode)

On mount, if `articleId !== null`, fetch the existing article:

```typescript
useEffect(() => {
  if (articleId) {
    loadArticle(articleId);
  }
}, [articleId]);

async function loadArticle(id: string) {
  setIsLoading(true);
  try {
    const res = await apiFetch(`/api/cms/articles/${id}`);
    if (!res.ok) {
      router.push("/admin/articles");
      return;
    }
    const { article } = await res.json();
    
    setTitle(article.title ?? "");
    setSlug(article.slug ?? "");
    setExcerpt(article.excerpt ?? "");
    setContent(article.content ?? "");
    setStatus(article.status ?? "draft");
    setFeaturedImageUrl(article.featured_image_url ?? "");
    setImages(article.article_images ?? []);
    setArticleId(article.id);
  } catch {
    router.push("/admin/articles");
  } finally {
    setIsLoading(false);
  }
}
```

## 2.4 Auto-Slug from Title

Debounce 500ms after title changes. Slug is auto-generated if the slug field is empty OR if it was last auto-generated (tracked via `slugAutoGenerated` ref).

```typescript
const slugAutoGeneratedRef = useRef(false);

useEffect(() => {
  const timer = setTimeout(() => {
    if (title && (!slug || slugAutoGeneratedRef.current)) {
      const auto = title
        .toLowerCase()
        .replace(/[^a-z0-9\s-]/g, "")
        .replace(/\s+/g, "-")
        .replace(/-+/g, "-")
        .replace(/^-|-$/g, "");
      setSlug(auto);
      slugAutoGeneratedRef.current = true;
    }
  }, 500);

  return () => clearTimeout(timer);
}, [title]);
```

If user manually edits the slug, set `slugAutoGeneratedRef.current = false`.

## 2.5 Autosave (30-Second Inactivity)

```typescript
// On any field change — mark dirty and reset timer
function markDirty() {
  setIsDirty(true);
  setLastSavedAt(null);  // clear "saved X ago" until next save
  
  if (autosaveTimer) clearTimeout(autosaveTimer);
  
  const timer = setTimeout(() => {
    if (isDirty && articleId) {
      performAutosave();
    }
  }, 30_000);  // 30 seconds
  
  setAutosaveTimer(timer);
}

// performAutosave — silent save, no UX feedback except "Saved X sec ago"
async function performAutosave() {
  if (!articleId || isSaving) return;
  
  await saveArticle({ autosave: true });
}
```

**Important:** `autosave: true` suppresses `published_at` changes on the server side.

## 2.6 Save Functions

### `saveArticle` — Core Save Logic

```typescript
interface SaveOptions {
  autosave?: boolean;   // true = silent, no toast, no dirty reset
  newStatus?: "draft" | "published" | "archived";  // override status
}

async function saveArticle(options: SaveOptions = {}) {
  if (!title.trim()) {
    setSaveError("Judul artikel wajib diisi");
    return false;
  }
  
  setIsSaving(true);
  setSaveError(null);
  
  try {
    // ── Step 1: Ensure article exists (create placeholder if new) ──
    let currentArticleId = articleId;
    
    if (!currentArticleId) {
      const createRes = await apiFetch("/api/cms/articles/draft", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ title: title.trim() }),
      });
      
      if (!createRes.ok) {
        const err = await createRes.json();
        setSaveError(err.error || "Gagal membuat artikel");
        return false;
      }
      
      const { article_id } = await createRes.json();
      currentArticleId = article_id;
      setArticleId(article_id);
    }
    
    // ── Step 2: Update article metadata ──
    const updatePayload: Record<string, unknown> = {
      title: title.trim(),
      slug: slug || undefined,
      excerpt: excerpt.trim() || undefined,
      content: content.trim() || undefined,
      featured_image_url: featuredImageUrl || undefined,
      autosave: options.autosave ?? false,
    };
    
    if (options.newStatus) {
      updatePayload.status = options.newStatus;
    }
    
    const updateRes = await apiFetch(`/api/cms/articles/${currentArticleId}`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(updatePayload),
    });
    
    if (!updateRes.ok) {
      const err = await updateRes.json();
      setSaveError(err.error || "Gagal menyimpan artikel");
      return false;
    }
    
    const { article } = await updateRes.json();
    
    // ── Step 3: Sync images if any new uploads happened ──
    // (GalleryUploader handles this separately via batch endpoint)
    
    if (!options.autosave) {
      setIsDirty(false);
      setLastSavedAt(new Date());
      if (options.newStatus === "published") {
        toast.success("Artikel dipublikasi");
      } else {
        toast.success("Draf disimpan");
      }
    }
    
    return true;
  } catch {
    setSaveError("Terjadi kesalahan. Silakan coba lagi.");
    return false;
  } finally {
    setIsSaving(false);
  }
}
```

```tsx
<GalleryUploader
  articleId={articleId}
  tempArticleId={tempArticleId}
  images={images}
  onImagesChange={setImages}
  onDirty={markDirty}
/>
```

**Critical flow for new articles:**
- `GalleryUploader` detects `articleId === null`
- On first image upload, calls `POST /api/cms/articles/draft` internally
- Returns `article_id` → composer sets `articleId` and `tempArticleId`
- All subsequent uploads use the real `articleId`

## 2.9 Before Unload Guard

```typescript
useEffect(() => {
  const handleBeforeUnload = (e: BeforeUnloadEvent) => {
    if (!isDirty) return;
    
    // If this was a draft article that was never published, mark for cleanup
    if (!articleId || status === "draft") {
      // Flag that server should delete this draft on next cleanup run
      localStorage.setItem("draft_cleanup_id", articleId ?? "");
    }
    
    e.preventDefault();
    e.returnValue = "";  // Chrome requires returnValue to be set
  };
  
  window.addEventListener("beforeunload", handleBeforeUnload);
  return () => window.removeEventListener("beforeunload", handleBeforeUnload);
}, [isDirty, articleId, status]);
```

**Note:** `localStorage` flag approach is a simple way to handle draft cleanup. For production, consider a cleanup API that runs on app init to purge stale drafts (>24h).

## 2.14 Edge Cases

| Scenario | Handling |
|----------|----------|
| User opens composer, types title, closes tab | `beforeunload` fires → draft flagged for cleanup |
| User creates article, uploads images, never saves | Images orphaned in R2 — cleanup job handles via `DELETE /api/cms/articles/{id}/images` |
| User saves draft, refreshes page | `articleId` lost — list view shows article in draft state (user must re-open) |
| Slug auto-generation conflicts with existing slug | Server returns 409 on PATCH → show error, user edits slug manually |
| Publish with no title | Blocked — show "Judul wajib diisi" error inline |
| Save while image upload in progress | Block `saveArticle` until all uploads complete (track via `uploadingCount` state) |
